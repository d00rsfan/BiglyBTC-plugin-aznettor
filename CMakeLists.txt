cmake_minimum_required(VERSION 3.25)
project(BiglyBTC-plugin-aznettor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL 3.5.5 EXACT REQUIRED COMPONENTS Crypto)
add_library(biglybtc_tor_auth STATIC safecookie.cpp)
target_include_directories(biglybtc_tor_auth PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(biglybtc_tor_auth PRIVATE OpenSSL::Crypto)
target_compile_features(biglybtc_tor_auth PUBLIC cxx_std_23)
set_target_properties(biglybtc_tor_auth PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(biglybtc_tor_auth PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(biglybtc-tor-auth-driver auth_driver.cpp)
target_link_libraries(biglybtc-tor-auth-driver PRIVATE biglybtc_tor_auth)
target_compile_options(biglybtc-tor-auth-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME original-tor-safecookie COMMAND Python3::Interpreter
  "${CMAKE_CURRENT_SOURCE_DIR}/fixture_test.py" "$<TARGET_FILE:biglybtc-tor-auth-driver>")
set_tests_properties(original-tor-safecookie PROPERTIES TIMEOUT 90 LABELS "unit;fixtures")
add_executable(biglybtc-tor-auth-bounds-tests auth_bounds_test.cpp)
target_link_libraries(biglybtc-tor-auth-bounds-tests PRIVATE biglybtc_tor_auth)
target_compile_options(biglybtc-tor-auth-bounds-tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bounded-tor-safecookie COMMAND biglybtc-tor-auth-bounds-tests)
set_tests_properties(bounded-tor-safecookie PROPERTIES TIMEOUT 30 LABELS "unit")
